import { Injectable } from '@angular/core';
import { HttpClient, HttpParams } from '@angular/common/http';
import { Observable } from 'rxjs';
import { environment } from '../../../../environments/environment';
import { DocumentDto, CategoryComplianceDto, DashboardSummary, WorkflowStageDto } from '../models/models';

@Injectable({ providedIn: 'root' })
export class DocumentService {
  private base = `${environment.apiUrl}/documents`;

  constructor(private http: HttpClient) {}

  // Dashboard summary
  getDashboardSummary(): Observable<DashboardSummary> {
    return this.http.get<DashboardSummary>(`${this.base}/dashboard/summary`);
  }

  // Category compliance list
  getCategoryCompliance(): Observable<CategoryComplianceDto[]> {
    return this.http.get<CategoryComplianceDto[]>(`${this.base}/categories/compliance`);
  }

  // Recent documents with server-side paging / filters
  getRecentDocuments(page = 1, pageSize = 10, search?: string, categoryId?: number, statusId?: number) {
    let params = new HttpParams().set('page', page.toString()).set('pageSize', pageSize.toString());
    if (search) params = params.set('search', search);
    if (categoryId) params = params.set('categoryId', categoryId.toString());
    if (statusId) params = params.set('statusId', statusId.toString());
    return this.http.get<{ items: DocumentDto[]; total: number }>(`${this.base}/recent`, { params });
  }

  // Get single document
  getDocument(documentId: number) {
    return this.http.get<DocumentDto>(`${this.base}/${documentId}`);
  }

  // Create (upload) document - expects FormData with file
  uploadDocument(formData: FormData) {
    return this.http.post<{ documentId: number }>(`${this.base}/upload`, formData);
  }

  // Update document metadata
  updateDocument(documentId: number, payload: Partial<DocumentDto>) {
    return this.http.put(`${this.base}/${documentId}`, payload);
  }

  // Update status / workflow action
  updateStatus(documentId: number, statusId: number, comments?: string) {
    return this.http.post(`${this.base}/${documentId}/status`, { statusId, comments });
  }

  // Workflow stages summary
  getWorkflowStages(): Observable<WorkflowStageDto[]> {
    return this.http.get<WorkflowStageDto[]>(`${this.base}/workflow/stages`);
  }

  // Bulk actions
  bulkReview(documentIds: number[]) {
    return this.http.post(`${this.base}/workflow/bulk-review`, { documentIds });
  }
}

import { Component, OnInit } from '@angular/core';
import { DocumentService } from '../../services/document.service';
import { DocumentDto } from '../../models/models';
import { FormControl } from '@angular/forms';
import { debounceTime } from 'rxjs/operators';

@Component({
  selector: 'app-recent-documents',
  templateUrl: './recent-documents.component.html'
})
export class RecentDocumentsComponent implements OnInit {
  documents: DocumentDto[] = [];
  total = 0;
  page = 1;
  pageSize = 5; // as requested
  search = new FormControl('');
  selectedCategory?: number;
  selectedStatus?: number;
  loading = false;

  constructor(private svc: DocumentService) {}

  ngOnInit(): void {
    this.load();

    this.search.valueChanges.pipe(debounceTime(400)).subscribe(() => {
      this.page = 1;
      this.load();
    });
  }

  load() {
    this.loading = true;
    this.svc.getRecentDocuments(this.page, this.pageSize, this.search.value, this.selectedCategory, this.selectedStatus)
      .subscribe({
        next: (res) => { this.documents = res.items; this.total = res.total; this.loading = false; },
        error: () => { this.loading = false; }
      });
  }

  onPageChange(newPage: number) {
    this.page = newPage;
    this.load();
  }

  clearFilters() {
    this.selectedCategory = undefined;
    this.selectedStatus = undefined;
    this.search.setValue('');
    this.load();
  }

  download(doc: DocumentDto) {
    // open download endpoint
    window.open(`${(location.origin)}${'/api/documents/' + doc.documentId + '/download'}`);
  }

  review(doc: DocumentDto) {
    // navigate to review screen or open modal (not included)
  }
}

import { Component } from '@angular/core';
import { DocumentService } from '../../services/document.service';

@Component({
  selector: 'app-document-upload',
  templateUrl: './document-upload.component.html'
})
export class DocumentUploadComponent {
  file?: File;
  title = '';
  version = '';
  categoryId?: number;

  constructor(private svc: DocumentService) {}

  onFileChange(event: any) {
    const f = event.target.files?.[0];
    if (f) this.file = f;
  }

  upload() {
    if (!this.file) return alert('Select file');
    const fd = new FormData();
    fd.append('file', this.file);
    fd.append('title', this.title);
    fd.append('version', this.version);
    if (this.categoryId) fd.append('categoryId', this.categoryId.toString());

    this.svc.uploadDocument(fd).subscribe({ next: (r) => { alert('Uploaded'); }, error: () => alert('Upload failed') });
  }
}

# Instructions
1. Copy the `src/app/modules/documents` folder into your Angular project.
2. Register `DocumentsModule` inside your `AppRoutingModule` with lazy loading if desired:
```ts
{ path: 'documents', loadChildren: () => import('./modules/documents/documents.module').then(m => m.DocumentsModule) }
```
3. Ensure `HttpClientModule` is imported in `AppModule` and `environment.apiUrl` points to your backend.
4. Install Tailwind or adapt SCSS to your styles. The components use simple utility classes.
